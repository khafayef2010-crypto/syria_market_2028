/**
 * دالة ذكية لمعالجة النص المستخرج بالـ OCR وتحليله إلى أرقام شراء ومبيع
 */
function parseRates(text) {
    const clean = text.replace(/,/g, '').replace(/،/g, '');
    const rates = [];

    // البحث عن أرقام الدولار (مثال: 15150 - 15250 أو 15200)
    const dollarMatches = clean.match(/(\d{5})/g);
    if (dollarMatches && dollarMatches.length >= 1) {
        const p1 = parseFloat(dollarMatches[0]);
        const p2 = dollarMatches.length > 1 ? parseFloat(dollarMatches[1]) : p1 + 100;
        rates.push({
            currency_name: 'USD (دولار أمريكي)',
            buy_price: Math.min(p1, p2),
            sell_price: Math.max(p1, p2)
        });
    }

    // البحث عن سعر غرام الذهب عيار 21 (بين 700,000 و 2,500,000)
    const goldMatches = clean.match(/(\d{6,7})/g);
    if (goldMatches) {
        for (const num of goldMatches) {
            const val = parseFloat(num);
            if (val >= 700000 && val <= 2500000) {
                rates.push({
                    currency_name: 'غرام الذهب عيار 21',
                    buy_price: val - 15000,
                    sell_price: val
                });
                break;
            }
        }
    }

    return rates;
}

/**
 * 2. استلام ومعالجة الصور القادمة للبوت أو المعاد توجيهها من القناة
 */
bot.on('photo', async (msg) => {
    const chatId = msg.chat.id;
    try {
        console.log(`📸 [${new Date().toLocaleTimeString('ar-SY')}] تم استلام صورة نشرة جديدة من: ${chatId}`);
        bot.sendMessage(chatId, '⏳ جاري استلام الصورة وتحليل أرقام الأسعار عبر محرك OCR الذكي...');

        // اختيار الصورة بأعلى دقة متوفرة
        const photo = msg.photo[msg.photo.length - 1];
        const fileLink = await bot.getFileLink(photo.file_id);
        console.log(`🔗 رابط ملف الصورة: ${fileLink}`);

        // قراءة النصوص والأرقام عبر Tesseract OCR (دعم اللغتين العربية والإنجليزية)
        const worker = await createWorker('ara+eng');
        const ret = await worker.recognize(fileLink);
        const extractedText = ret.data.text;
        await worker.terminate();

        console.log('📝 النص المستخرج من الصورة:\n' + extractedText);

        // استخراج وتنسيق أسعار الصرف والذهب
        const parsedRates = parseRates(extractedText);
        const rowsToInsert = [];

        if (parsedRates.length > 0) {
            for (const r of parsedRates) {
                rowsToInsert.push({
                    currency_name: r.currency_name,
                    buy_price: r.buy_price,
                    sell_price: r.sell_price,
                    raw_text: extractedText,
                    image_url: fileLink
                });
            }
        } else {
            rowsToInsert.push({
                currency_name: 'نشرة عامة (ALSRAFEN)',
                buy_price: null,
                sell_price: null,
                raw_text: extractedText,
                image_url: fileLink
            });
        }

        // حفظ وتحديث البيانات في جدول exchange_rates في Supabase
        const { data, error } = await supabase
            .from('exchange_rates')
            .insert(rowsToInsert)
            .select();

        if (error) {
            console.error('❌ خطأ أثناء الإدخال في Supabase:', error.message);
            throw error;
        }

        let responseMsg = '✅ تم قراءة النشرة وتحديث أسعار الصرف في التطبيق بنجاح!\n\n';
        if (parsedRates.length > 0) {
            parsedRates.forEach(r => {
                responseMsg += `💵 ${r.currency_name}: شراء ${r.buy_price?.toLocaleString()} | مبيع ${r.sell_price?.toLocaleString()} ل.س\n`;
            });
        }
        responseMsg += '\n🇸🇾 تظهر الآن في شريط بورصة التطبيق للمستخدمين فوراً.';

        bot.sendMessage(chatId, responseMsg);
        console.log('✅ تم تحديث ونشر الأسعار في Supabase بنجاح!');

    } catch (err) {
        console.error('❌ حدث خطأ أثناء المعالجة:', err);
        bot.sendMessage(chatId, `⚠️ تعذر إتمام التحديث: ${err.message || err}`);
    }
});
